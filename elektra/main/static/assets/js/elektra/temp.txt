
## API за теми и подтеми по специалност

### 1. Добавяне на нови изгледи във views.py

Добавете следните функции във вашия съществуващ `views.py` файл:

```python
@api_view(['GET'])
def specialty_themes_view(request, specialty_id):
    """
    Изглед, който връща всички теми и подтеми за зададена специалност.
    Използва съществуващите сериализатори без да ги променя.
    """
    try:
        # Проверяваме дали специалността съществува
        specialty = get_object_or_404(Specialty, id=specialty_id)

        # Намираме всички теми за тази специалност
        themes = Theme.objects.filter(specialty=specialty).order_by('num')

        # Общ резултат, който ще върнем
        result = {
            'specialty': {
                'id': specialty.id,
                'professional_field_num': specialty.professional_field_num,
                'professional_field_name': specialty.professional_field_name,
                'profession_num': specialty.profession_num,
                'profession_name': specialty.profession_name,
                'specialty_num': specialty.specialty_num,
                'specialty_name': specialty.specialty_name,
                'nip': request.build_absolute_uri(specialty.nip.url) if specialty.nip else ""
            },
            'themes': []
        }

        # За всяка тема извличаме нейните точки (подтеми)
        for theme in themes:
            # Сериализирайте темата с ThemeSerializer
            theme_serializer = ThemeSerializer(theme)
            theme_data = theme_serializer.data

            # Намираме всички точки (подтеми) за тази тема
            theme_items = ThemeItem.objects.filter(theme_id=theme).order_by('item')

            # Сериализирайте точките с ThemeItemSerializer
            theme_items_serializer = ThemeItemSerializer(theme_items, many=True)

            # Добавяме точките към темата
            theme_data['items'] = theme_items_serializer.data

            # Добавяме темата с нейните точки към резултата
            result['themes'].append(theme_data)

        return Response(result)

    except Exception as e:
        print(f"Грешка при извличане на теми за специалност: {str(e)}")
        import traceback
        traceback.print_exc()
        return Response(
            {'error': str(e)},
            status=status.HTTP_500_INTERNAL_SERVER_ERROR
        )


@api_view(['GET'])
def theme_detail_view(request, theme_id):
    """
    Изглед, който връща детайли за конкретна тема и нейните подтеми.
    """
    try:
        # Проверяваме дали темата съществува
        theme = get_object_or_404(Theme, id=theme_id)

        # Сериализираме темата
        theme_serializer = ThemeSerializer(theme)
        theme_data = theme_serializer.data

        # Намираме всички точки (подтеми) за тази тема
        theme_items = ThemeItem.objects.filter(theme_id=theme).order_by('item')

        # Сериализираме точките
        theme_items_serializer = ThemeItemSerializer(theme_items, many=True)

        # Добавяме точките към темата
        theme_data['items'] = theme_items_serializer.data

        return Response(theme_data)

    except Exception as e:
        print(f"Грешка при извличане на детайли за тема: {str(e)}")
        import traceback
        traceback.print_exc()
        return Response(
            {'error': str(e)},
            status=status.HTTP_500_INTERNAL_SERVER_ERROR
        )
```

### 3. Пример за JavaScript код за извикване на API

Ето пример как да извикате новото API от вашия JavaScript/Vue код:

```javascript
// Метод за зареждане на теми и подтеми за специалност
loadThemesForSpecialty() {
    const vm = this;
    vm.isLoading = true;

    axios.get('/api/specialty/' + SPEC + '/themes/')
        .then(function(response) {
            vm.specialty = response.data.specialty;
            vm.themes = response.data.themes;
            vm.isLoading = false;
        })
        .catch(function(error) {
            console.error("Грешка при зареждане на теми:", error);
            vm.error = "Възникна грешка при зареждане на темите.";
            vm.isLoading = false;
        });
},

// Метод за зареждане на детайли за конкретна тема
loadThemeDetails(themeId) {
    const vm = this;
    vm.isLoadingTheme = true;

    axios.get('/api/themes/' + themeId + '/')
        .then(function(response) {
            vm.currentTheme = response.data;
            vm.themeItems = response.data.items;
            vm.isLoadingTheme = false;
        })
        .catch(function(error) {
            console.error("Грешка при зареждане на детайли за тема:", error);
            vm.themeError = "Възникна грешка при зареждане на детайлите за темата.";
            vm.isLoadingTheme = false;
        });
}
```

### 4. Разширяване на API с функции за управление на теми и подтеми

Ако искате да добавите API за създаване, редактиране и изтриване на теми и подтеми, можете да добавите следните функции:

```python
@api_view(['POST'])
def create_theme_view(request):
    """
    Изглед за създаване на нова тема.
    """
    try:
        # Вземаме данните от заявката
        specialty_id = request.data.get('specialty')
        specialty = get_object_or_404(Specialty, id=specialty_id)

        # Определяме номера на новата тема
        num = request.data.get('num')
        if not num:
            # Ако не е предоставен номер, намираме максималния номер и увеличаваме с 1
            max_num = Theme.objects.filter(specialty=specialty).aggregate(models.Max('num'))['num__max'] or 0
            num = max_num + 1

        # Създаваме новата тема
        theme = Theme.objects.create(
            specialty=specialty,
            num=num,
            title=request.data.get('title', ''),
            tasks_total=request.data.get('tasks_total', 24),
            tasks_knowledge=request.data.get('tasks_knowledge', 0),
            tasks_comprehension=request.data.get('tasks_comprehension', 0),
            tasks_application=request.data.get('tasks_application', 0),
            tasks_analysis=request.data.get('tasks_analysis', 0)
        )

        # Сериализираме и връщаме новата тема
        serializer = ThemeSerializer(theme)
        return Response(serializer.data, status=status.HTTP_201_CREATED)

    except Exception as e:
        print(f"Грешка при създаване на тема: {str(e)}")
        import traceback
        traceback.print_exc()
        return Response(
            {'error': str(e)},
            status=status.HTTP_500_INTERNAL_SERVER_ERROR
        )

@api_view(['PUT', 'DELETE'])
def update_delete_theme_view(request, theme_id):
    """
    Изглед за обновяване или изтриване на тема.
    """
    try:
        theme = get_object_or_404(Theme, id=theme_id)

        if request.method == 'PUT':
            # Обновяваме темата
            theme.title = request.data.get('title', theme.title)
            theme.num = request.data.get('num', theme.num)
            theme.tasks_total = request.data.get('tasks_total', theme.tasks_total)
            theme.tasks_knowledge = request.data.get('tasks_knowledge', theme.tasks_knowledge)
            theme.tasks_comprehension = request.data.get('tasks_comprehension', theme.tasks_comprehension)
            theme.tasks_application = request.data.get('tasks_application', theme.tasks_application)
            theme.tasks_analysis = request.data.get('tasks_analysis', theme.tasks_analysis)
            theme.save()

            # Сериализираме и връщаме обновената тема
            serializer = ThemeSerializer(theme)
            return Response(serializer.data)

        elif request.method == 'DELETE':
            # Изтриваме темата
            theme.delete()
            return Response(status=status.HTTP_204_NO_CONTENT)

    except Exception as e:
        print(f"Грешка при обновяване/изтриване на тема: {str(e)}")
        import traceback
        traceback.print_exc()
        return Response(
            {'error': str(e)},
            status=status.HTTP_500_INTERNAL_SERVER_ERROR
        )
```

### Важна забележка относно TaskSerializer

В оригиналния код на ThemeItemSerializer забелязвам, че включвате TaskSerializer:

```python
class ThemeItemSerializer(serializers.ModelSerializer):
    tasks = TaskSerializer(many=True)

    class Meta:
        model = ThemeItem
        fields = "__all__"
```

Тъй като не искате да променяте съществуващите сериализатори, аз запазих тази структура в моето решение. Ако TaskSerializer и връзката между ThemeItem и Task не са релевантни за новото API, данните за tasks просто ще бъдат включени в JSON отговора.
